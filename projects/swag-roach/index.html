using UnityEngine;
using UnityEngine.InputSystem;
using static UnityEngine.UI.Image;

[RequireComponent(typeof(Rigidbody2D))]
[RequireComponent(typeof(BoxCollider2D))]
public class Player : MonoBehaviour {
Rigidbody2D rb;
BoxCollider2D hitbox;
public static Player instance;

[SerializeField] LayerMask groundLayer;
[SerializeField] private GameObject dashEffect;
[SerializeField] private float maxSpeed = 23f;
[SerializeField] private float jumpSpeed = 20f;
[SerializeField] private float jumpBufferTime = 0.12f;
[SerializeField] private float jumpCoyoteTime = 0.16f;

[SerializeField] private float airDamping = 0.8f;
[SerializeField] private float airRotationDamping = 0.75f;
[SerializeField] private float groundDamping = 2.5f;
[SerializeField] private float groundRotationDamping = 12f;
[SerializeField] private float jumpSpinSpeed = 12f;
[SerializeField] private float jumpSpinThreshold = 15f;
[SerializeField] private float jumpSpeedMult = 1.03f;


[SerializeField] private float gravityHangThreshold = 2f;
[SerializeField] private float regularGravity = 6.5f;
[SerializeField] private float hangGravity = 3f;
[SerializeField] private float gravityBoost = 3f;

[SerializeField] private float dashDuration = 3f;
[SerializeField] private float dashSpeed = 10f;
[SerializeField] private float dashAdjustSpeed = 0.1f;
[SerializeField] private float dashGravity = 0.4f;

[SerializeField] private float boosterSpeed = 20f;

[SerializeField] private GameObject particlePrefab;

public bool dashUnlocked = false;
Animator animator;

public Vector2 moveDirection = Vector2.zero;
private float _lastJumpBuffered = -Mathf.Infinity ;
private bool _dashHeld;
private bool _canDash;
public bool jumped;
private float _dashStartedTime;
private Vector2 _dashDirection;

public bool IsMoving() {
return moveDirection.magnitude > 0;
}

public bool IsGrounded() {
Vector2 dir = new Vector2(Mathf.Cos((rb.rotation - 90) * Mathf.Deg2Rad), Mathf.Sin((rb.rotation - 90) * Mathf.Deg2Rad));
RaycastHit2D groundCheck = Physics2D.Raycast(rb.position, dir, 0.75f, groundLayer);
Debug.DrawLine(rb.position, rb.position + dir * 0.8f, Color.red);
return groundCheck.collider != null;
}

public Vector2 GetGroundNormal() {
float rayOffset = 0;
if(Mathf.Abs(rb.linearVelocityX) > 7) {
rayOffset = Mathf.Sign(rb.linearVelocityX) * 0.04f;
}
RaycastHit2D groundCheck = Physics2D.Raycast(rb.position + rayOffset * Vector2.right, Vector2.down, 0.8f, groundLayer);
Debug.DrawRay(rb.position + rayOffset * 0.3f * Vector2.right, Vector2.down);
if (groundCheck.collider != null)
return groundCheck.normal;
else
return Vector2.zero;
}

public void Awake() {
instance = this;
rb = GetComponent<Rigidbody2D>();
animator = GetComponent<Animator>();
hitbox = GetComponent<BoxCollider2D>();
}
private void FixedUpdate() {
if (IsGrounded()) {
ProcessWalk();
}
ProcessJump();
ProcessFriction();
ProcessGravity();
if (dashUnlocked)
ProcessDash();

ProcessAnimations();
ProcessParticles();
}

private void ProcessAnimations() {
animator.SetBool("pumping", moveDirection.y < 0);
}
bool everyother = false;
private void ProcessParticles() {
if (IsGrounded() && rb.linearVelocity.magnitude > 15) {
if (everyother) {
var g = Instantiate(particlePrefab);
g.transform.position = hitbox.bounds.min;
}
everyother = !everyother;
}

}


public void BufferJump() {
_lastJumpBuffered = Time.time;
}
public void Jump() {
rb.freezeRotation = false;
_dashStartedTime = -Mathf.Infinity;
jumped = true;
rb.linearVelocityY = Mathf.Max(rb.linearVelocityY, 0);
rb.linearVelocityX *= jumpSpeedMult;
if(rb.linearVelocity.magnitude > jumpSpinThreshold) {
rb.angularVelocity = jumpSpinSpeed;
}
rb.AddForce(jumpSpeed * Vector2.up);

SFXManager.instance.PlaySound(SFXManager.instance.jump);
}
public void CutJump() {
if(rb.linearVelocityY > 0)
rb.linearVelocityY /= 2;
}
private bool isPumping() {
if(moveDirection.y < 0 && GetGroundNormal()==Vector2.zero) return true;
return (moveDirection.y < 0 && Mathf.Sign(GetGroundNormal().x) == Mathf.Sign(rb.linearVelocityX));
}

private void ProcessGravity() {
if (Mathf.Abs(rb.linearVelocityY) < gravityHangThreshold) {
rb.gravityScale = hangGravity;
}
else {
rb.gravityScale = regularGravity;
}
if (isPumping()) {
if (!IsGrounded() && GetGroundNormal() == Vector2.zero) {
rb.rotation = Mathf.LerpAngle(rb.rotation, 0, 0.5f);
}
rb.gravityScale += gravityBoost;
GetComponentInChildren<TrailRenderer>().emitting = true;
}
else {
GetComponentInChildren<TrailRenderer>().emitting = false;
}
}
private void OnCollisionStay2D(Collision2D collision) {
if (collision.gameObject.tag.Equals("Ground")) {
ProcessGround();
}
}
private void OnTriggerEnter2D(Collider2D collision) {
if (collision.tag.Equals("Booster")) {
rb.linearVelocity = Vector2.zero;
rb.AddForce(collision.transform.right * boosterSpeed);
CameraShake.instance.Shake(0.5f, 0.2f);
SFXManager.instance.PlaySound(SFXManager.instance.boost);
}
}
private void ProcessGround() {

Vector2 groundNormal = new Vector2(-GetGroundNormal().y, GetGroundNormal().x);
if (rb.freezeRotation = (groundNormal != Vector2.zero && !isJumping) ) {
if(isPumping()) rb.AddForce(-GetGroundNormal() * 0.3f + 0.2f*Mathf.Sign(rb.linearVelocityX) * Vector2.right);
Debug.Log(GetGroundNormal());
rb.rotation = Mathf.LerpAngle(rb.rotation,Mathf.Atan(groundNormal.y / groundNormal.x) * Mathf.Rad2Deg, 0.67f);
//rb.AddForce(GetGroundNormal());
}
}
private void ProcessFriction() {
if (IsGrounded()) {
rb.linearDamping = groundDamping;
rb.angularDamping = groundRotationDamping;
}
else {
rb.linearDamping = airDamping;
rb.angularDamping = airRotationDamping;
}
}
float lastGroundedTime = -Mathf.Infinity;
bool isJumping;
private void ProcessJump() {
if (IsGrounded()) lastGroundedTime = Time.time;
if (isJumping = (Time.time < _lastJumpBuffered + jumpBufferTime && Time.time < lastGroundedTime + jumpCoyoteTime)) {
_lastJumpBuffered = -Mathf.Infinity;
lastGroundedTime = -Mathf.Infinity;
Jump();
}
}
private void ProcessWalk() {
float targetSpeed = moveDirection.x * maxSpeed;

if (Mathf.Abs(rb.linearVelocityX) < maxSpeed) {
rb.AddForce(new Vector2(targetSpeed, 0f), ForceMode2D.Force);
}
}

public bool IsDashing() {
return _dashHeld && Time.time < _dashStartedTime + dashDuration;
}
private void ProcessDash() {
if (IsGrounded()) _canDash = true;

if (_dashHeld) {
dashEffect.transform.position = transform.position;
if (Time.time < _dashStartedTime + dashDuration) {
rb.linearDamping = groundDamping;
rb.AddForce(moveDirection * dashSpeed);
rb.gravityScale = -dashGravity;
Instantiate(dashEffect, rb.position, Quaternion.identity);
}
else {
_canDash = false;
}
}
else {
_dashStartedTime = -Mathf.Infinity;
}

}


private void StartDash() {
_dashDirection = moveDirection;
_dashStartedTime = Time.time;
_dashHeld = true;
_canDash = false;
AuraCounter.instance.GainAura(500, "spraypaint flying");
CutJump();
}
public void InputDash(InputAction.CallbackContext context) {
if (context.performed && _canDash) {
StartDash();


}
else if (context.canceled) {
_dashHeld = false;

}
}
public void InputJump(InputAction.CallbackContext context) {
if (context.performed) {
BufferJump();

}
if (context.canceled && jumped) {
CutJump();
jumped = false;
}
}
public void InputMove(InputAction.CallbackContext context) {
moveDirection = context.ReadValue<Vector2>();
}
}
